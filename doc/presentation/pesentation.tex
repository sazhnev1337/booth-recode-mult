% ── СЛАЙД 2: Актуальность ───────────────────────────────────
\begin{frame}{Актуальность}
  \begin{itemize}
    \item Умножитель — \textbf{доминирующий} блок по площади и энергопотреблению в DSP
    \item В адаптивных фильтрах работает непрерывно $\Rightarrow$ определяет энергобюджет
    \item Многие DSP-задачи \textbf{устойчивы к погрешности} $\Rightarrow$ приближённые вычисления:
          \\[2pt]
          \hspace{1em}\alert{доли \% точности} $\longleftrightarrow$ \alert{ощутимая экономия энергии}
    \item Базовый приём — кодирование \textbf{Бута radix-4} (вдвое меньше частичных произведений)
  \end{itemize}

  \vfill
  \begin{block}{Ключевой вопрос}
    Как применить \textbf{recoding с zero-skipping}
    (Университет Бэйхан, 2025~\cite{zhu2025})
    в реалистичном сценарии \textbf{адаптивного фильтра}?
  \end{block}
\end{frame}

% ── СЛАЙД 3: Цель и задачи ──────────────────────────────────
\begin{frame}{Цель и задачи}
  \textbf{Цель работы:} снизить энергопотребление умножителя Бута
  в адаптивном цифровом фильтре за счёт перекодировки с пропуском
  нулевых кодов.

  \vskip 12pt
  \textbf{Задачи:}
  \begin{enumerate}
    \item Адаптировать алгоритм recoding (Университет Бэйхан~\cite{zhu2025})
          к \textbf{параллельной} архитектуре умножителя
    \item Предложить архитектуру с \alert{вынесенным наружу} перекодировщиком
          (амортизация по тапам фильтра)
    \item Реализовать на Verilog три варианта: \texttt{naive}, \texttt{booth},
          \texttt{booth\_extrec}
    \item Построить методику измерения площади и мощности
          (открытый маршрут синтеза, 45\,нм)
    \item Оценить точность приближённых режимов (NMED, MRED, SNR)
          на статистике $10^6$ пар
  \end{enumerate}
\end{frame}

      \textbf{Умножитель Бута radix-4} в составе
      \textbf{$N$-тапового адаптивного фильтра}

      \vskip 8pt
      \textbf{Принцип radix-4 Booth:}
      \begin{itemize}
        \item Множитель — перекрывающиеся группы по 3 бита
        \item Каждая группа $\to$ код из набора $\{\bar 2,\bar 1,0,1,2\}$

        Коэффициенты обновляются \alert{редко} (раз в $10$--$10^3$ тактов),
        поток отсчётов — \alert{каждый такт}
      \end{block}

% ── СЛАЙД 5: Метод перекодировки (zero-skipping) ────────────
\begin{frame}{Метод: перекодировка с пропуском нулей}
  \begin{columns}[T]
    \column{0.52\textwidth}
      Идея (Университет Бэйхан~\cite{zhu2025}): пара соседних
      ненулевых кодов Бута переписывается в эквивалентную пару,
      \alert{содержащую ноль}:
      \[
        K_{2i+1}=M_{2i+1}-\overline{M_{2i+1}},\quad
        K_{2i}=M_{2i}+4M_{2i+1}
      \]
      \begin{itemize}
        \item \textbf{Точно} (ER): паттерны \texttt{00100}, \texttt{11011},
              ошибка $\mathrm{ED}=0$
        \item \textbf{Приближённо} (AR): ещё 4 паттерна
              (\texttt{00101}, \texttt{00110}, \texttt{11001}, \texttt{11010}),
              $\mathrm{ED}=\pm1$
      \end{itemize}
      \vskip 4pt
      Больше нулевых кодов $\Rightarrow$ больше \textbf{занулённых
      частичных произведений}.

    \column{0.46\textwidth}
      % РИСУНОК: dot-diagram PP-матрицы.
      % Показать: слева обычный Booth (все строки заполнены),
      % справа после recoding — часть строк зануляется (пустые точки).
      \centering
      \includegraphics[width=\textwidth]{zaglushka.pdf}
      \\[2pt]
      {\footnotesize Зануление строк в матрице частичных произведений}
  \end{columns}
\end{frame}

% ── СЛАЙД 6: Проблема прямого переноса ──────────────────────
\begin{frame}{Проблема прямого переноса в параллельную схему}
  \begin{columns}[T]
    \column{0.52\textwidth}
      В оригинале — \textbf{последовательный} умножитель:
      перекодировщик работает по одному такту на код,
      экономия — от \alert{пропуска тактов}.

      \vskip 8pt
      В параллельной схеме пропускать такты нечего —
      все частичные произведения формируются \textbf{одновременно}.
      Перекодировщик пришлось бы ставить в \alert{каждую строку}:

      \begin{itemize}
        \item рост площади и задержки
        \item накладные расходы съедают выигрыш от занулений
        \item арбитраж между строками усложняет схему
      \end{itemize}

      \vskip 4pt
      \begin{block}{Вывод}
        Нужен другой способ «встроить» перекодировку в параллельный
        умножитель.
      \end{block}

    \column{0.46\textwidth}
      % РИСУНОК: параллельный умножитель с перекодировщиком
      % в КАЖДОЙ строке PP (подсветить дублирование recoder'ов
      % красным — mipt-accent). Подпись "наивный перенос".
      \centering
      \includegraphics[width=\textwidth]{zaglushka.pdf}
      \\[2pt]
      {\footnotesize Перекодировщик дублируется в каждой строке}
  \end{columns}
\end{frame}

% ── СЛАЙД 7: Предлагаемая архитектура (главный слайд) ────────
\begin{frame}{Предлагаемая архитектура: вынесенная перекодировка}
  \begin{columns}[T]
    \column{0.46\textwidth}
      \textbf{Ключевая идея:} перекодировщик вынесен \alert{наружу}
      умножителя.

      \vskip 6pt
      \begin{itemize}
        \item В регистре умножителя хранится \textbf{уже перекодированный}
              вектор (24 бита: \texttt{two}, \texttt{one}, \texttt{neg})
        \item Перекодировка считается \alert{один раз} при обновлении
              коэффициента
        \item Стоимость \textbf{амортизируется} по $N$ тапам и по времени
        \item Умножитель — \textbf{один и тот же} RTL для всех режимов
              (точный / приближённые)
      \end{itemize}

    \column{0.52\textwidth}
      % РИСУНОК (ГЛАВНЫЙ): две схемы.

      % коэффициента), в умножитель идёт recoded-вектор.

% ── СЛАЙД 8: Реализация ─────────────────────────────────────
\begin{frame}{Реализация}
  \begin{columns}[T]
    \column{0.50\textwidth}
      \textbf{Три RTL-модуля (Verilog):}
      \begin{itemize}
        \item \texttt{mult\_naive} — прямое \texttt{a*b} (опорная точка)
        \item \texttt{mult\_booth} — radix-4 Booth, перекодировщик
              \textbf{внутри}
        \item \texttt{mult\_booth\_extrec} — приём \textbf{готового}
              24-битного вектора
      \end{itemize}

      \vskip 8pt
      \textbf{Перекодировщик — на Python:}
      \begin{itemize}
        \item последовательный алгоритм, все 7 паттернов
        \item генерирует вектор \texttt{\{two, one, neg\}}
        \item режимы: \texttt{exact}, \texttt{approx\_1..4}
      \end{itemize}
