% ── СЛАЙД 2: Актуальность ───────────────────────────────────
\begin{frame}{Актуальность}
  \begin{itemize}
    \item Умножитель — \textbf{доминирующий} блок по площади и энергопотреблению в DSP
    \item В адаптивных фильтрах работает непрерывно $\Rightarrow$ определяет энергобюджет
    \item Многие DSP-задачи \textbf{устойчивы к погрешности} $\Rightarrow$ приближённые вычисления:
          \\[2pt]
          \hspace{1em}\alert{доли \% точности} $\longleftrightarrow$ \alert{ощутимая экономия энергии}
    \item Базовый приём — кодирование \textbf{Бута radix-4} (вдвое меньше частичных произведений)
  \end{itemize}

  \vfill
  \begin{block}{Ключевой вопрос}
    Как применить метод с дополнительной перекодировкой
    (Университет Бэйхан, 2025~\cite{zhu2025})
    в реалистичном сценарии \textbf{адаптивного фильтра}?
  \end{block}
\end{frame}

% ── СЛАЙД 3: Цель и задачи ──────────────────────────────────
\begin{frame}{Цель и задачи}
  \textbf{Цель работы:} снизить энергопотребление умножителя Бута
  в адаптивном цифровом фильтре, применив методы перекодировки
  с добавлением нулевых кодов.

  \vskip 12pt
  \textbf{Задачи:}
  \begin{enumerate}
    \item Адаптировать алгоритм перекодировки~\cite{zhu2025}
          к \textbf{параллельной} архитектуре умножителя
    \item Предложить архитектурное решение, реализующее преимущества
          перекодировки в сценарии \textbf{адаптивного фильтра}
    \item Реализовать предложенную архитектуру на Verilog
    \item Построить методику измерения площади и мощности
          на \textbf{общедоступных} инструментах (45\,нм)
    \item Оценить точность приближённых режимов (NMED, MRED, SNR)
          на статистике $10^6$ пар
  \end{enumerate}
\end{frame}

% ── СЛАЙД 4: Объект исследования ────────────────────────────
\begin{frame}{Объект исследования}
  \begin{columns}[T]
    \column{0.50\textwidth}
      \textbf{Умножитель Бута radix-4} в составе
      \textbf{$N$-тапового адаптивного фильтра}

      \vskip 8pt
      \textbf{Принцип radix-4 Booth:}
      \begin{itemize}
        \item Множитель — перекрывающиеся группы по 3 бита
        \item Каждая группа $\to$ код из набора $\{\bar 2,\bar 1,0,1,2\}$ 3 бита: (One, Two, Neg)
        \item \textbf{Вдвое меньше} частичных произведений
      \end{itemize}

      \vskip 8pt
      \begin{block}{Ключевая асимметрия}
        Коэффициенты обновляются \alert{редко} (раз в $10$--$10^3$ тактов),
        поток отсчётов — \alert{каждый такт}
      \end{block}

    \column{0.50\textwidth}
      % РИСУНОК: N-тапповый адаптивный FIR.

\end{frame}

% ── СЛАЙД 5: Метод перекодировки (zero-skipping) ────────────
\begin{frame}{Метод: перекодировка с пропуском нулей}
  \begin{columns}[T]
    \column{0.52\textwidth}
      Идея (Университет Бэйхан~\cite{zhu2025}): пара соседних
      ненулевых кодов Бута переписывается в эквивалентную пару,
      \alert{содержащую ноль}:
      \[
        K_{2i+1}=M_{2i+1}-\overline{M_{2i+1}},\quad
        K_{2i}=M_{2i}+4M_{2i+1}
      \]
      \begin{itemize}
        \item \textbf{Точно} (ER): паттерны \texttt{00100}, \texttt{11011},
              ошибка $\mathrm{ED}=0$
        \item \textbf{Приближённо} (AR): ещё 4 паттерна
              (\texttt{00101}, \texttt{00110}, \texttt{11001}, \texttt{11010}),
              $\mathrm{ED}=\pm1$
      \end{itemize}
      \vskip 4pt
      Больше нулевых кодов $\Rightarrow$ больше \textbf{занулённых
      частичных произведений}.

    \column{0.46\textwidth}
      % РИСУНОК: dot-diagram PP-матрицы.
      % Показать: слева обычный Booth (все строки заполнены),
      % справа после recoding — часть строк зануляется (пустые точки).
      \centering
      \includegraphics[width=\textwidth]{recoding.pdf}
      \\[2pt]
      {\footnotesize }
  \end{columns}
\end{frame}

% ── СЛАЙД 6: Проблема прямого переноса ──────────────────────
\begin{frame}[shrink=10]{Проблема: последовательный умножитель в конвейере}
  \begin{columns}[T]
    \column{0.54\textwidth}
      В оригинале — \textbf{последовательный} умножитель:
      нулевые коды \alert{пропускаются}, число тактов переменно.

      \vskip 5pt
      FIR-фильтр требует результата \textbf{каждый такт}
      $\Rightarrow$ нужен контроллер готовности:

      \begin{itemize}
        \item добавляет площадь и динамическую мощность
        \item накладные расходы \textbf{перекрывают} выигрыш
        \item реальный DSP требует \textbf{фиксированной}
              пропускной способности
      \end{itemize}

      \vskip 4pt
      \begin{block}{Вывод}
        Необходим параллельный умножитель с \textbf{фиксированной
        однотактной} задержкой.
      \end{block}

    \column{0.44\textwidth}
      \centering
      \vskip 40pt
      \includegraphics[width=\textwidth]{problem.pdf}
  \end{columns}
\end{frame}

% ── СЛАЙД 7: Проблема: перекодировка внутри умножителя ───────
\begin{frame}{Проблема: перекодировка внутри умножителя}
  \begin{columns}[T]
    \column{0.48\textwidth}
      Алгоритм перекодировки обрабатывает пары кодов Бута
      \alert{последовательно}, слева направо:

      \vskip 8pt
      \begin{itemize}
        \item Пары \textbf{перекрываются} $\Rightarrow$ каждая зависит
              от предыдущей $\Rightarrow$ \textbf{нельзя распараллелить}
        \item Перекодировщик — дополнительная комбинационная логика
              \textbf{внутри} умножителя $\Rightarrow$ сам потребляет энергию
      \end{itemize}

      \vskip 10pt
      Простое добавление перекодировки в параллельный умножитель
      делает модуль \textbf{только хуже}.

    \column{0.50\textwidth}
      \centering
      \vskip 20pt
      % РИСУНОК: схема — перекодировщик внутри умножителя,
      % показать последовательную цепочку зависимых пар.
        \includegraphics[width=\textwidth]{sequential.pdf}%
  \end{columns}
\end{frame}

% ── СЛАЙД 8: Предлагаемая архитектура (главный слайд) ────────
\begin{frame}{Предлагаемая архитектура: вынесенная перекодировка}
  \begin{columns}[T]
    \column{0.44\textwidth}
      \textbf{Ключевая идея:} перекодировщик вынесен \alert{наружу}
      умножителя.

      \vskip 6pt
      \begin{itemize}
        \item В регистре умножителя хранится \textbf{уже перекодированный}
              вектор (24 бита: \texttt{two}, \texttt{one}, \texttt{neg})
        \item Перекодировка считается \alert{один раз} при обновлении
              коэффициента
        \item Стоимость \textbf{амортизируется} по $N$ тапам и по времени
        \item Умножитель — \textbf{один и тот же} RTL для всех режимов
              (точный / приближённые)
      \end{itemize}

    \column{0.56\textwidth}
      % РИСУНОК (ГЛАВНЫЙ): две схемы.
      % Сверху — baseline: recoder ВНУТРИ умножителя.
      % Снизу — предлагаемая: recoder СНАРУЖИ (в блоке обновления
      % коэффициента), в умножитель идёт recoded-вектор.
      % Выделить вынесенный recoder цветом mipt-accent.
      \centering
      \includegraphics[width=\textwidth]{extrec.pdf}
      \\[2pt]
      {\footnotesize Слева — базовая схема, справа — с вынесенной
      перекодировкой}
  \end{columns}
\end{frame}

% ── СЛАЙД 9: Реализация ─────────────────────────────────────
\begin{frame}{Реализация}
  \textbf{Два RTL-модуля (Verilog):}
  \begin{itemize}
    \item \texttt{mult\_booth} — \textbf{исходный}: radix-4 Booth,
          перекодировщик \textbf{внутри} (эталон для сравнения)
    \item \texttt{mult\_booth\_extrec} — \textbf{предлагаемый}: принимает
          \textbf{готовый} 24-битный вектор, перекодировщика нет
  \end{itemize}
  \vskip 10pt
  \textbf{Перекодировщик — на Python:}
  \begin{itemize}
    \item последовательный алгоритм, обрабатывает все 7 пар кодов Бута
    \item генерирует вектор \texttt{\{two, one, neg\}} для умножителя
    \item \textbf{exact} — точный: нулевая ошибка, только 2 паттерна
    \item \textbf{approx\_1..4} — приближённые: каждый следующий уровень
          добавляет паттерны с $\mathrm{ED}{=}\pm1$, агрессивность растёт
  \end{itemize}
\end{frame}

% ── СЛАЙД 10: Площадь ───────────────────────────────────────
\begin{frame}{Результаты: площадь}
  \centering
  \renewcommand{\arraystretch}{1.4}
  \begin{tabular}{lr}
    \toprule
    Модуль & Площадь, мкм² \\
    \midrule
    \texttt{mult\_booth}         & 1523 \\
    \texttt{mult\_booth\_extrec} & 1518 \\
    \bottomrule
  \end{tabular}

  \vskip 14pt
  \begin{block}{Два компенсирующих фактора}
    \textbf{$-$} Из умножителя убрана вся логика кодирования Бута
    (вычисление \texttt{neg}, \texttt{one}, \texttt{two} из коэффициента)\\[4pt]
    \textbf{$+$} Входной регистр коэффициента расширен:
    16 бит $\to$ 24 бита (перекодированный вектор)\\[4pt]
    Два эффекта взаимно компенсируются $\Rightarrow$
    площадь \textbf{практически не меняется}
  \end{block}
\end{frame}

% ── СЛАЙД 11: Мощность ──────────────────────────────────────
\begin{frame}{Результаты: мощность}
  \begin{columns}[T]
    \column{0.56\textwidth}
      \textbf{Сценарий gauss\_slow} (самый реалистичный):
      \vskip 4pt
      \small\renewcommand{\arraystretch}{1.25}
      \begin{tabular}{lrr}
        \toprule
        Конфигурация & Мощность, мкВт & Экономия \\
        \midrule
        booth (базовый)  & 245              & ---       \\
        extrec exact     & 244              & $+0.4\%$  \\
        extrec approx\_1 & 241              & $+1.6\%$  \\
        extrec approx\_2 & \textbf{238}     & \alert{$+2.9\%$} \\
        \bottomrule
      \end{tabular}

      \vskip 8pt
      \begin{block}{Почему gauss/slow}
        \footnotesize
        \textbf{Slow:} коэффициент стабилен дольше, перекодировка
        амортизируется на большем числе тактов\\[3pt]
        \textbf{Gauss:} реалистичный сценарий — речевой и аудиосигнал
        гауссово распределён; абсолютная экономия одинакова для обоих
        распределений ($\approx 7$\,мкВт)
      \end{block}

    \column{0.46\textwidth}
      \centering
      \includegraphics[width=\textwidth]{power_dut_comparison_gauss_slow.png}
  \end{columns}
\end{frame}

% ── СЛАЙД 13: Компромисс мощность / точность ────────────────
\begin{frame}{Компромисс: мощность vs точность}
  \centering\small
  \renewcommand{\arraystretch}{1.4}
  \begin{tabular}{lcc}
    \toprule
    Режим & Экономия мощн. & SNR (гаусс) \\
    \midrule
    standard  & $-1.6\%$          & $\infty$    \\
    exact     & $+0.4\%$          & $\infty$    \\

  \end{tabular}

  \vskip 10pt
  \begin{block}{Оценка approx\_2 для практики}
    SNR = 52\,дБ $\gg$ 40\,дБ  ---
    безопасен для многих DSP-приложений.
    Смещение $\approx 0$ исключает накопление ошибки.
  \end{block}
\end{frame}

% ── СЛАЙД 15: Заключение ────────────────────────────────────
\begin{frame}{Заключение}
  \begin{itemize}
    \item Вынесенный рекодер: \textbf{$-2.9\%$ мощности}
          (gauss\_slow, approx\_2)
    \item Площадь умножителя \textbf{не возрастает} — два эффекта
          компенсируют друг друга
    \item Единый RTL для всех режимов точности
    \item approx\_2: SNR $= 52\,$дБ, смещение $\approx 0$ —
          пригоден для большинства DSP-задач
  \end{itemize}

  \vskip 10pt
  \begin{block}{Главный результат}
    Перекодировщик вынесен наружу и \textbf{усложнён} методом
    дополнительной перекодировкой: он создаёт нулевые
    частичные произведения до того, как данные попадают
    в умножитель. Сам умножитель при этом \textbf{упрощается}
    — его площадь не растёт, а мощность снижается.
  \end{block}
\end{frame}
